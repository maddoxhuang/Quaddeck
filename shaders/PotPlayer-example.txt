// PotPlayer / MPC-style pixel shader. QuadDeck converts this legacy syntax to D3D11.
sampler s0 : register(s0);
float4 p0 : register(c0);
float4 p1 : register(c1);

#define width (p0[0])
#define height (p0[1])
#define px (p1[0])
#define py (p1[1])

float4 main(float2 tex : TEXCOORD0) : COLOR
{
    float4 center = tex2D(s0, tex) * 1.8;
    center -= tex2D(s0, tex + float2(px, 0)) * 0.2;
    center -= tex2D(s0, tex - float2(px, 0)) * 0.2;
    center -= tex2D(s0, tex + float2(0, py)) * 0.2;
    center -= tex2D(s0, tex - float2(0, py)) * 0.2;
    return float4(saturate(center.rgb), 1.0);
}
